cmake_minimum_required(VERSION 3.19)
project(sega_representation_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
get_filename_component(SEGA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
if(NOT BACKEND OR NOT UPSTREAM)
  message(FATAL_ERROR "Specify -DBACKEND=megahit|bifrost|sbwt|sshash|metagraph and -DUPSTREAM=/source/path")
endif()
add_executable(repr_probe probe.cpp)
target_compile_definitions(repr_probe PRIVATE REPR_HEADER="${BACKEND}.hpp")
target_include_directories(repr_probe PRIVATE ${SEGA_ROOT}/include ${SEGA_ROOT}/bench ${SEGA_ROOT}/backends)
if(BACKEND STREQUAL "megahit")
  find_package(Threads REQUIRED)
  target_sources(repr_probe PRIVATE ${UPSTREAM}/src/sdbg/sdbg_meta.cpp ${UPSTREAM}/src/sdbg/sdbg_raw_content.cpp)
  target_include_directories(repr_probe PRIVATE ${UPSTREAM}/src)
  target_compile_definitions(repr_probe PRIVATE SDBGT_WITH_MEGAHIT)
  target_link_libraries(repr_probe PRIVATE Threads::Threads)
elseif(BACKEND STREQUAL "bifrost")
  set(COMPILATION_ARCH OFF CACHE STRING "No native CPU target" FORCE)
  set(ENABLE_AVX2 OFF CACHE BOOL "Portable Bifrost build" FORCE)
  add_subdirectory(${UPSTREAM} upstream)
  set_target_properties(bifrost_static bifrost_dynamic Bifrost PROPERTIES CXX_STANDARD 14)
  target_compile_definitions(repr_probe PRIVATE MAX_KMER_SIZE=32 MAX_GMER_SIZE=32)
  target_link_libraries(repr_probe PRIVATE bifrost_static)
elseif(BACKEND STREQUAL "sshash")
  add_subdirectory(${UPSTREAM} upstream)
  target_link_libraries(repr_probe PRIVATE sshash_static)
  # Upstream headers use the same x86 intrinsics as their library.
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    target_compile_options(repr_probe PRIVATE -mbmi2 -mavx2)
  endif()
elseif(BACKEND STREQUAL "sbwt")
  if(APPLE)
    add_compile_definitions(_LIBCPP_ENABLE_CXX17_REMOVED_RANDOM_SHUFFLE)
  endif()
  set(GIT_SUBMODULE OFF CACHE BOOL "Sources are pinned before configuration" FORCE)
  add_subdirectory(${UPSTREAM} upstream)
  foreach(legacy KMC_CORE_OBJS KMC_TOOLS_OBJS KMC_API_OBJS KMC_CLI_OBJS)
    set_target_properties(${legacy} PROPERTIES CXX_STANDARD 14)
  endforeach()
  target_include_directories(repr_probe PRIVATE ${UPSTREAM}/include/sbwt ${UPSTREAM}/sdsl-lite/include
    ${UPSTREAM}/KMC ${UPSTREAM}/SeqIO/include ${CMAKE_CURRENT_BINARY_DIR}/upstream/include
    ${CMAKE_CURRENT_BINARY_DIR}/upstream/external/sdsl-lite/build/external/libdivsufsort/include)
  target_link_libraries(repr_probe PRIVATE sbwt_static)
elseif(BACKEND STREQUAL "metagraph")
  set(METAGRAPH_BUILD_EXECUTABLE OFF CACHE BOOL "Use the pinned CLI package for construction" FORCE)
  add_subdirectory(${UPSTREAM}/metagraph upstream)
  # Conda's transitive binary dependencies can also install Folly headers.
  # The source library must use its pinned Folly headers consistently.
  foreach(t metagraph-core folly)
    target_include_directories(${t} SYSTEM BEFORE PUBLIC
      ${UPSTREAM}/metagraph/external-libraries/folly
      ${CMAKE_CURRENT_BINARY_DIR}/upstream/external-libraries/folly/include)
  endforeach()
  target_include_directories(metagraph-core SYSTEM BEFORE PUBLIC
    ${UPSTREAM}/metagraph/external-libraries/sdsl-lite/include)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    # SIMDe's emulated AVX vector ABI warnings must not make the portable
    # implementation unbuildable. Keep the warnings in the build log.
    file(GLOB alignment_sources ${UPSTREAM}/metagraph/src/graph/alignment/*.cpp)
    set_source_files_properties(${alignment_sources} DIRECTORY ${UPSTREAM}/metagraph
      PROPERTIES COMPILE_OPTIONS "-Wno-error=psabi")
    set_source_files_properties(${UPSTREAM}/metagraph/src/common/vectors/vector_algorithm.cpp
      DIRECTORY ${UPSTREAM}/metagraph PROPERTIES COMPILE_OPTIONS "-Wno-error=psabi")
  endif()
  target_compile_features(repr_probe PRIVATE cxx_std_20)
  target_link_libraries(repr_probe PRIVATE metagraph-core)
else()
  message(FATAL_ERROR "Unsupported BACKEND")
endif()
